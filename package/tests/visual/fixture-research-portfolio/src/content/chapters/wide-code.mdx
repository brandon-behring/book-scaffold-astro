under the docs-style left sidebar. The guard (`layout-overflow.spec.ts`) renders
this chapter at 1024 / 1280 / 1440px and asserts the **page** never scrolls

def fit_hierarchical_model(observations, priors, *, max_iter=1000, tol=1e-8):
    """Two-level hierarchical Bayesian model via coordinate-ascent VI."""
    posterior = initialise_from_prior(priors, jitter=1e-6, seed=20260621)
    for iteration in range(max_iter):
        expected_stats = compute_expectations(observations, posterior, normalize=True)
        posterior = update_variational_parameters(expected_stats, priors, damping=0.5)
        if has_converged(posterior, tol=tol) and iteration > 10:  # warmup before early exit
            return posterior, {"iterations": iteration, "converged": True, "final_tolerance": tol}
    return posterior, {"iterations": max_iter, "converged": False, "note": "hit max_iter without meeting the convergence tolerance — inspect priors"}
